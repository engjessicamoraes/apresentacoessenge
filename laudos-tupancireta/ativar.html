Passo 1 – Copiar o código 
/**
 * Recebimento dos laudos de unidades habitacionais – Tupanciretã (SENGE Solidário).
 * Cada envio do site grava o PDF na pasta e atualiza uma linha da planilha.
 */
const PASTA_ID = "1qbm5UGT0vDlUSCLykX5iVSN1oTir__JX";
const PLANILHA_ID = "1mVM42dsrL3pWRwEOUUtqqCX8F3Z0Kf4x4fNU7_Ac-BI";
const CHAVE = "senge-solidario-tupancireta-2026";
const CAB = ["ID do laudo","Recebido em","Responsável/família","CPF","Endereço","Bairro/localidade","Coordenadas","Mapa","Tipo construtivo","Moradores","Danos","Habitabilidade","Valor estimado (R$)","Responsável técnico","Registro","Nº de fotos","PDF"];

function doGet() {
  return resp({ ok: true, servico: "laudos-uh", mensagem: "Recebimento de laudos ativo." });
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  try {
    const d = JSON.parse(e.postData.contents);
    if (d.chave !== CHAVE) return resp({ ok: false, erro: "chave" });
    lock.waitLock(30000);
    const pasta = DriveApp.getFolderById(PASTA_ID);
    const sub = subpasta(pasta, "PDFs dos laudos");
    // substitui a versão anterior do mesmo laudo
    const it = sub.getFiles();
    while (it.hasNext()) { const f = it.next(); if (f.getDescription() === "laudo:" + d.id) f.setTrashed(true); }
    const blob = Utilities.newBlob(Utilities.base64Decode(d.pdf), "application/pdf", d.nomePdf);
    const arq = sub.createFile(blob).setDescription("laudo:" + d.id);
    const m = d.meta || {};
    const linha = [d.id, new Date(), m.familia, m.cpf, m.endereco, m.bairro, m.coord, m.mapa, m.tipo, m.moradores, m.danos, m.hab, m.valor, m.rt, m.registro, m.fotos, arq.getUrl()];
    const sh = SpreadsheetApp.openById(PLANILHA_ID).getSheets()[0];
    if (sh.getRange(1, 1).getValue() !== CAB[0]) { sh.clear(); sh.getRange(1, 1, 1, CAB.length).setValues([CAB]).setFontWeight("bold"); sh.setFrozenRows(1); }
    const ids = sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues().map(r => String(r[0])) : [];
    const i = ids.indexOf(String(d.id));
    if (i >= 0) sh.getRange(i + 2, 1, 1, linha.length).setValues([linha]);
    else sh.appendRow(linha);
    return resp({ ok: true, pdf: arq.getUrl() });
  } catch (err) {
    return resp({ ok: false, erro: String(err) });
  } finally {
    try { lock.releaseLock(); } catch (x) {}
  }
}

function subpasta(p, nome) { const it = p.getFoldersByName(nome); return it.hasNext() ? it.next() : p.createFolder(nome); }
function resp(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

/** Execute uma vez pelo editor para autorizar o acesso ao Drive e à planilha. */
function autorizar() { DriveApp.getFolderById(PASTA_ID).getName(); SpreadsheetApp.openById(PLANILHA_ID).getName(); }
Passo 2 – Criar o script Abrir novo script do Google
  1. No menu ⋮ do Chrome, marque “Versão para computador” (no iPhone: botão aA › “Solicitar site para computador”).
  2. Toque dentro do código que aparece, pressione e segure › Selecionar tudo › Colar.
  3. Toque no ícone de disquete (Salvar).
Passo 3 – Autorizar
  1. Na barra de cima, ao lado de “Executar”, escolha a função autorizar e toque em Executar.
  2. Revisar permissões › escolha a sua conta › Avançado › Acessar projeto (não seguro) › Permitir. O aviso aparece porque o script é seu e não foi publicado na loja do Google.
Passo 4 – Publicar
  1. Implantar › Nova implantação › na engrenagem, escolha App da Web.
  2. Executar como: Eu · Quem pode acessar: Qualquer pessoa › Implantar.
  3. Copie a URL do app da Web (termina em /exec) e envie para o Claude.
